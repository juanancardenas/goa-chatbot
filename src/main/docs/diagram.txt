package domain {

  package "model" as corePackage {
    class Conversation {
      -String id
      -String userId
      -String engagementLetterId
      -ConversationStatus status
      -ConversationType type
      -LocalDateTime createdAt
      -Integer lastSequenceNumber
    }

    class Message {
      -String id
      -String conversationId
      -MessageSenderType senderType
      -MessageType messageType
      -String content
      -LocalDateTime timestamp
      -Integer sequenceNumber
      -String parentMessageId
    }

    class Escalation {
      -UUID id
      -String conversationId
      -String userId
      -LocalDateTime createdAt
      -String phone
      -String email
    }
  }

  package "model.chatbot" as chatbotPackage {

    package "command" as commandPackage {
      class ChatbotMessageCommand {
        -String conversationId
        -String message
      }

      class ChatbotContextualConversationCommand {
        -String engagementLetterId
      }
    }

    package "reply" as replyPackage {
      class ChatbotReplyDecision {
        -String assistantReply
        -ChatbotResponseMode responseMode
        -boolean usedPlatformData
        -List<String> sourcesSummary
      }
    }

    package "result" as resultPackage {
      class ChatbotConfigurationResult {
        -boolean enabled
        -String provider
        -String model
        -int maxInputCharacters
        -int maxOutputTokens
        -int maxContextMessages
        -boolean documentsAvailable
      }

      class ChatbotContextualConversationResult {
        -String conversationId
        -String engagementLetterId
        -String createdAt
        -String error
      }

      class ChatbotMessageResult {
        -String conversationId
        -String message
        -String error
        -String createdAt
        -ChatbotResponseMode responseMode
        -Boolean usedPlatformData
        -List<String> sourcesSummary
      }

      class ChatbotConversationHistoryResult {
        -String conversationId
        -String engagementLetterId
        -String type
        -String status
        -Integer page
        -Integer size
        -Boolean hasMore
        -Long totalMessages
        -List<ChatbotHistoryMessageResult> messages
      }

      class ChatbotHistoryMessageResult {
        -String id
        -String conversationId
        -String senderType
        -String messageType
        -String content
        -String timestamp
        -Integer sequenceNumber
        -String parentMessageId
      }

      class ChatbotConversationSummaryResult {
        -String conversationId
        -String type
        -String status
        -String engagementLetterId
        -String createdAt
        -String lastMessageAt
        -String preview
      }
    }
  }

  package "model.ai" as aiPackage {
    class ChatbotAiRequest {
      -String conversationId
      -String userId
      -String userMessage
      -String basePrompt
      -String roleProfile
      -String conversationType
      -String platformContext
      -List<String> recentMessages
      -String model
      -Integer maxOutputTokens
      -Double temperature
      -Boolean documentsAvailable
    }

    class ChatbotAiResponse {
      -String content
      -String provider
      -String model
      -String finishReason
      -Integer usedTokens
      -String error
    }
  }

  package "model.security" as securityPackage {
    class AuthenticatedUserContext {
      -String userId
      -ConversationProfileType profile
    }
  }

  package "model.pagination" as paginationPackage {
    class "PageResult<T>" as PageResult {
      -List<T> content
      -int page
      -int size
      -boolean hasNext
      -long totalElements
    }
  }

  package "model.platform" as platformPackage {
    class ChatbotDocumentContext {
      -boolean available
      -boolean authorizedSourceConfigured
      -List<String> visibleDocumentTitles
      -List<String> sourcesSummary
    }

    class ChatbotPlatformContext {
      -String engagementLetterId
      -String ownerDisplayName
      -List<String> procedureTitles
      -List<String> legalTaskSummaries
      -List<String> recentEventSummaries
      -List<String> sourcesSummary
    }

    class EngagementLetterSummary {
      -UUID id
      -LocalDate creationDate
      -LocalDate closingDate
      -UserSummary owner
      -List<LegalProcedureSummary> legalProcedures
    }

    class LegalProcedureSummary {
      -String title
      -LocalDate startDate
      -LocalDate closingDate
      -List<String> legalTasks
    }

    class EngagementEventPage {
      -List<EngagementEventSummary> content
    }

    class EngagementEventSummary {
      -String type
      -String state
      -String title
      -String comment
      -LocalDate date
      +String displayText()
    }

    class UserSummary {
      -UUID id
      -String firstName
      -String familyName
      -String email
      -String mobile
      +String displayName()
    }
  }

  package "enums" as enumsPackage {
    enum ConversationType {
      GENERAL
      CONTEXTUAL
    }

    enum ConversationStatus {
      ACTIVE
      CLOSED
      ARCHIVED
    }

    enum MessageSenderType {
      USER
      ASSISTANT
      SYSTEM
    }

    enum MessageType {
      REQUEST
      RESPONSE
      INSTRUCTION
    }

    enum ConversationProfileType {
      CLIENT
      PROFESSIONAL
    }

    enum ChatbotResponseMode {
      GENERAL
      CONTEXTUAL_PLATFORM_DATA
      CONTEXTUAL_RESTRICTED
    }

    enum ChatbotAiProvider {
      OLLAMA
      OPENAI
      GEMINI
      +isSupported(value: String): boolean
    }

    enum ChatbotScopeViolationReason {
      OUT_OF_CASE_SCOPE
      MISSING_CASE_CONTEXT
      LEGAL_BINDING_ADVICE_REQUESTED
      EMOTIONAL_DISTRESS
      UNSUPPORTED_FACTUAL_ASSERTION
      AMBIGUOUS_CONTEXT
      OUT_OF_DOMAIN
    }

    enum PlatformQuestionType {
      ENGAGEMENT_STATUS
      LEGAL_TASKS
      TIMELINE_EVENTS
      DOCUMENTS
      GENERAL_CONTEXT
    }
  }
}

package "external microservices" as externalPackage {
  class "goa-user\nUser" as ExternalUser <<external>> {
    +UUID id
  }

  class "goa-engagement\nEngagementLetter" as ExternalEngagementLetter <<external>> {
    +UUID id
  }
}
